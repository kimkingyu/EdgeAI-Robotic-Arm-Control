cmake_minimum_required(VERSION 3.16)
project(EdgeAI_Robotic_Arm_Control LANGUAGES CXX C)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()

option(EDGEAI_VISION_ONLY "Build standalone vision/tests without the arm executable" OFF)
option(EDGEAI_BUILD_YOLO_VISION "Build independent YOLO vision executable" ON)
option(EDGEAI_VISION_WITH_RKNN "Explicitly enable the real RKNN C API" OFF)
set(EDGEAI_RKNN_INCLUDE_DIR "" CACHE PATH "Pinned official SDK directory containing rknn_api.h")
set(EDGEAI_RKNN_LIBRARY "" CACHE FILEPATH "Matching deployed librknnrt.so")

if(CMAKE_SYSTEM_PROCESSOR MATCHES "aarch64")
    add_definitions(-DARCH_AARCH64)
endif()
find_package(Threads REQUIRED)
if(EDGEAI_VISION_ONLY)
    find_package(OpenCV QUIET COMPONENTS core imgproc imgcodecs videoio)
else()
    find_package(OpenCV REQUIRED)
endif()
include(CTest)

# Preserve the existing arm target and its defaults. The vision-only build
# does not compile/link an actuator or initialize an I2C device.
if(NOT EDGEAI_VISION_ONLY)
    include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include
                        ${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/rknn_api/include
                        ${OpenCV_INCLUDE_DIRS})
    link_directories(${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/rknn_api/lib)
    add_executable(edge_arm_control src/main.cpp src/kinematics.cpp)
    set_target_properties(edge_arm_control PROPERTIES
        RUNTIME_OUTPUT_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/bin)
    target_compile_options(edge_arm_control PRIVATE -O3 -Wall -fopenmp)
    target_link_libraries(edge_arm_control PRIVATE ${OpenCV_LIBS} Threads::Threads)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/3rdparty/rknn_api/lib/librknnrt.so")
        target_link_libraries(edge_arm_control PRIVATE rknnrt)
    endif()
    if(BUILD_TESTING AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
        add_executable(test_cpp_control tools/test_cpp_control.cpp src/kinematics.cpp)
        target_compile_options(test_cpp_control PRIVATE -Wall -Wextra -Wpedantic)
        target_link_options(test_cpp_control PRIVATE
            "LINKER:--wrap=open" "LINKER:--wrap=ioctl" "LINKER:--wrap=read"
            "LINKER:--wrap=write" "LINKER:--wrap=close" "LINKER:--wrap=usleep"
            "LINKER:--wrap=__read_chk")
        add_test(NAME cpp_control_offline COMMAND test_cpp_control)
        add_executable(handeye_dump tools/handeye_dump.cpp)
        target_compile_options(handeye_dump PRIVATE -Wall -Wextra)
        target_link_libraries(handeye_dump PRIVATE ${OpenCV_LIBS})
    endif()
endif()

if(EDGEAI_BUILD_YOLO_VISION)
    add_library(yolo_vision STATIC src/vision/yolo_vision.cpp src/vision/yolo_rknn_runtime.cpp)
    target_include_directories(yolo_vision PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
    target_link_libraries(yolo_vision PUBLIC Threads::Threads)
    if(OpenCV_FOUND)
        target_compile_definitions(yolo_vision PUBLIC EDGEAI_HAVE_OPENCV=1)
        target_include_directories(yolo_vision PUBLIC ${OpenCV_INCLUDE_DIRS})
        target_link_libraries(yolo_vision PUBLIC ${OpenCV_LIBS})
    else()
        target_compile_definitions(yolo_vision PUBLIC EDGEAI_HAVE_OPENCV=0)
        message(STATUS "OpenCV unavailable: pure geometry/tests only; inference refuses to run")
    endif()
    if(EDGEAI_VISION_WITH_RKNN)
        if(NOT EXISTS "${EDGEAI_RKNN_INCLUDE_DIR}/rknn_api.h" OR NOT EXISTS "${EDGEAI_RKNN_LIBRARY}")
            message(FATAL_ERROR "Explicit real RKNN build requires pinned header and runtime library")
        endif()
        if(NOT OpenCV_FOUND)
            message(FATAL_ERROR "Real RKNN image pipeline requires OpenCV development files")
        endif()
        target_compile_definitions(yolo_vision PUBLIC EDGEAI_HAVE_RKNN=1)
        target_include_directories(yolo_vision PRIVATE ${EDGEAI_RKNN_INCLUDE_DIR})
        target_link_libraries(yolo_vision PUBLIC ${EDGEAI_RKNN_LIBRARY})
    else()
        target_compile_definitions(yolo_vision PUBLIC EDGEAI_HAVE_RKNN=0)
    endif()
    add_executable(edge_yolo_vision src/yolo_vision_main.cpp)
    target_link_libraries(edge_yolo_vision PRIVATE yolo_vision)
    set_target_properties(edge_yolo_vision PROPERTIES RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/bin)
    if(BUILD_TESTING)
        add_executable(test_yolo_vision tools/test_yolo_vision.cpp)
        target_link_libraries(test_yolo_vision PRIVATE yolo_vision)
        add_test(NAME yolo_vision_offline COMMAND test_yolo_vision)
    endif()
endif()
